```swift
import Foundation
import Sodium
import Clibsodium  // C API for crypto_pwhash (Argon2id)

/// Errors from cryptographic operations
enum CryptoError: Error, LocalizedError {
    case decryptionFailed
    case invalidCiphertext(String)
    case invalidKeySize(Int)
    case unsupportedParallelism(String)
    case argon2Failed

    var errorDescription: String? {
        switch self {
        case .decryptionFailed:
            return "Decryption failed — wrong key or tampered data"
        case .invalidCiphertext(let msg):
            return "Invalid ciphertext: \(msg)"
        case .invalidKeySize(let size):
            return "Key must be 32 bytes, got \(size)"
        case .unsupportedParallelism(let msg):
            return msg
        case .argon2Failed:
            return "Argon2id key derivation failed"
        }
    }
}

/// Swift crypto layer matching the TypeScript @noble/ciphers binary format.
/// Read/decrypt only — no encrypt operations.
struct CryptoConstants {
    static let keySize = 32       // 256-bit
    static let nonceSize = 24     // XChaCha20 nonce
    static let tagSize = 16       // Poly1305 tag
    static let saltSize = 16      // Argon2id salt
    static let overhead = 40      // nonceSize + tagSize
}

struct CryptoBridge {
    private static let sodium = Sodium()

    /// Decrypt XChaCha20-Poly1305 ciphertext in the managed-nonce format.
    /// Input format: [24-byte nonce][ciphertext][16-byte tag]
    /// Returns decrypted plaintext.
    static func decrypt(_ ciphertext: Data, key: Data) throws -> Data {
        guard key.count == CryptoConstants.keySize else {
            throw CryptoError.invalidKeySize(key.count)
        }
        guard ciphertext.count > CryptoConstants.overhead else {
            throw CryptoError.invalidCiphertext(
                "Too short: \(ciphertext.count) bytes, need > \(CryptoConstants.overhead)"
            )
        }

        let nonce = ciphertext.prefix(CryptoConstants.nonceSize)
        let encryptedWithTag = ciphertext.dropFirst(CryptoConstants.nonceSize)

        // libsodium expects: [ciphertext][tag] with separate nonce
        guard let decrypted = sodium.aead.xchacha20poly1305ietf.decrypt(
            authenticatedCipherText: Bytes(encryptedWithTag),
            secretKey: Bytes(key),
            nonce: Bytes(nonce)
        ) else {
            throw CryptoError.decryptionFailed
        }

        return Data(decrypted)
    }

    /// Unwrap a DEK using a KEK via XChaCha20-Poly1305.
    /// Input: wrappedDEK (72 bytes = 24 nonce + 32 encrypted + 16 tag), KEK (32 bytes)
    /// Returns: 32-byte DEK
    static func unwrapDEK(_ wrappedDEK: Data, kek: Data) throws -> Data {
        let dek = try decrypt(wrappedDEK, key: kek)
        guard dek.count == CryptoConstants.keySize else {
            throw CryptoError.invalidCiphertext(
                "Unwrapped DEK is \(dek.count) bytes, expected \(CryptoConstants.keySize)"
            )
        }
        return dek
    }

    /// Derive a KEK from a PIN/password using Argon2id.
    /// Only supports p=1 (libsodium limitation).
    static func deriveKEK(
        password: String,
        salt: Data,
        params: Argon2Params
    ) throws -> Data {
        guard params.p == 1 else {
            throw CryptoError.unsupportedParallelism(
                "libsodium only supports p=1; vault uses p=\(params.p)"
            )
        }
        guard salt.count == CryptoConstants.saltSize else {
            throw CryptoError.invalidCiphertext("Salt must be \(CryptoConstants.saltSize) bytes")
        }

        var passwordBytes = Array(password.utf8)
        defer {
            // Zero password bytes after derivation (matches TypeScript kdf.ts behavior)
            for i in passwordBytes.indices { passwordBytes[i] = 0 }
        }
        var output = Data(count: params.dkLen)

        let result = output.withUnsafeMutableBytes { outputPtr in
            salt.withUnsafeBytes { saltPtr in
                crypto_pwhash(
                    outputPtr.baseAddress!.assumingMemoryBound(to: UInt8.self),
                    UInt64(params.dkLen),
                    passwordBytes, UInt64(passwordBytes.count),
                    saltPtr.baseAddress!.assumingMemoryBound(to: UInt8.self),
                    UInt64(params.t),         // opslimit = t (time cost)
                    Int(params.m) * 1024,     // memlimit = m * 1024 (KiB → bytes)
                    Int32(crypto_pwhash_ALG_ARGON2ID13)
                )
            }
        }

        guard result == 0 else {
            throw CryptoError.argon2Failed
        }

        return output
    }
}

/// Argon2id parameters (matches TypeScript Argon2Params type)
struct Argon2Params {
    let t: Int      // time cost (iterations)
    let m: Int      // memory cost (KiB)
    let p: Int      // parallelism (must be 1 for libsodium)
    let dkLen: Int  // derived key length (bytes)
}
```

```swift
import Foundation

/// Parsed vault header matching the TypeScript VaultHeader type.
struct VaultHeader {
    let version: UInt8
    let masterSalt: Data
    let recoverySalt: Data
    let argon2Params: Argon2Params
    let masterWrappedDEK: Data
    let recoveryWrappedDEK: Data
}

/// Parses the binary vault header format.
/// Format: [1B version][16B masterSalt][16B recoverySalt]
///         [4B t LE][4B m LE][4B p LE][4B dkLen LE]
///         [2B masterWrappedDEK.len LE][...bytes]
///         [2B recoveryWrappedDEK.len LE][...bytes]
struct VaultHeaderParser {
    enum ParseError: Error, LocalizedError {
        case tooShort(Int)
        case unsupportedVersion(UInt8)
        case truncated(String)

        var errorDescription: String? {
            switch self {
            case .tooShort(let size):
                return "Vault header too short: \(size) bytes"
            case .unsupportedVersion(let v):
                return "Unsupported vault version: \(v)"
            case .truncated(let field):
                return "Vault header truncated at \(field)"
            }
        }
    }

    /// Parse a base64-encoded vault header string from Keychain.
    static func parse(base64String: String) throws -> VaultHeader {
        guard let data = Data(base64Encoded: base64String) else {
            throw ParseError.tooShort(0)
        }
        return try parse(data: data)
    }

    /// Parse raw vault header bytes.
    static func parse(data: Data) throws -> VaultHeader {
        // Minimum: 1 + 16 + 16 + 16 + 2 + 2 = 53 bytes
        guard data.count >= 53 else {
            throw ParseError.tooShort(data.count)
        }

        var offset = 0

        // Version (1 byte)
        let version = data[offset]
        guard version == 1 else {
            throw ParseError.unsupportedVersion(version)
        }
        offset += 1

        // Master salt (16 bytes)
        let masterSalt = data[offset..<offset + 16]
        offset += 16

        // Recovery salt (16 bytes)
        let recoverySalt = data[offset..<offset + 16]
        offset += 16

        // Argon2 params (4x uint32 LE)
        guard offset + 16 <= data.count else {
            throw ParseError.truncated("argon2Params")
        }
        let t = readUInt32LE(data, at: offset); offset += 4
        let m = readUInt32LE(data, at: offset); offset += 4
        let p = readUInt32LE(data, at: offset); offset += 4
        let dkLen = readUInt32LE(data, at: offset); offset += 4

        let argon2Params = Argon2Params(
            t: Int(t), m: Int(m), p: Int(p), dkLen: Int(dkLen)
        )

        // Master wrapped DEK (length-prefixed uint16 LE)
        guard offset + 2 <= data.count else {
            throw ParseError.truncated("masterWrappedDEK length")
        }
        let masterLen = Int(readUInt16LE(data, at: offset)); offset += 2
        guard offset + masterLen <= data.count else {
            throw ParseError.truncated("masterWrappedDEK data")
        }
        let masterWrappedDEK = data[offset..<offset + masterLen]
        offset += masterLen

        // Recovery wrapped DEK (length-prefixed uint16 LE)
        guard offset + 2 <= data.count else {
            throw ParseError.truncated("recoveryWrappedDEK length")
        }
        let recoveryLen = Int(readUInt16LE(data, at: offset)); offset += 2
        guard offset + recoveryLen <= data.count else {
            throw ParseError.truncated("recoveryWrappedDEK data")
        }
        let recoveryWrappedDEK = data[offset..<offset + recoveryLen]

        return VaultHeader(
            version: version,
            masterSalt: Data(masterSalt),
            recoverySalt: Data(recoverySalt),
            argon2Params: argon2Params,
            masterWrappedDEK: Data(masterWrappedDEK),
            recoveryWrappedDEK: Data(recoveryWrappedDEK)
        )
    }

    // MARK: - Little-endian helpers

    private static func readUInt16LE(_ data: Data, at offset: Int) -> UInt16 {
        return UInt16(data[offset]) | (UInt16(data[offset + 1]) << 8)
    }

    private static func readUInt32LE(_ data: Data, at offset: Int) -> UInt32 {
        return UInt32(data[offset])
            | (UInt32(data[offset + 1]) << 8)
            | (UInt32(data[offset + 2]) << 16)
            | (UInt32(data[offset + 3]) << 24)
    }
}
```

```swift
import Foundation
import SQLite3

/// Read-only access to the vault items SQLite database
/// in the App Group shared container.
struct DatabaseReader {
    enum DatabaseError: Error, LocalizedError {
        case notFound(String)
        case openFailed(String)
        case queryFailed(String)

        var errorDescription: String? {
            switch self {
            case .notFound(let path):
                return "Vault database not found at \(path)"
            case .openFailed(let msg):
                return "Failed to open database: \(msg)"
            case .queryFailed(let msg):
                return "Database query failed: \(msg)"
            }
        }
    }

    struct EncryptedItem {
        let id: String
        let type: String
        let encryptedDataBase64: String
    }

    /// Read all credential items from the shared SQLite database.
    /// Opens read-only to prevent WAL journal creation.
    static func readCredentials() throws -> [EncryptedItem] {
        let dbPath = try getDatabasePath()

        var db: OpaquePointer?
        let flags = SQLITE_OPEN_READONLY | SQLITE_OPEN_NOMUTEX
        let rc = sqlite3_open_v2(dbPath, &db, flags, nil)

        guard rc == SQLITE_OK, let database = db else {
            let msg = db.map { String(cString: sqlite3_errmsg($0)) } ?? "unknown"
            sqlite3_close(db)
            throw DatabaseError.openFailed(msg)
        }

        defer { sqlite3_close(database) }

        let query = "SELECT id, type, encrypted_data FROM vault_items WHERE type = 'credential'"
        var stmt: OpaquePointer?

        guard sqlite3_prepare_v2(database, query, -1, &stmt, nil) == SQLITE_OK else {
            let msg = String(cString: sqlite3_errmsg(database))
            throw DatabaseError.queryFailed(msg)
        }

        defer { sqlite3_finalize(stmt) }

        var items: [EncryptedItem] = []
        while sqlite3_step(stmt) == SQLITE_ROW {
            guard let idPtr = sqlite3_column_text(stmt, 0),
                  let typePtr = sqlite3_column_text(stmt, 1),
                  let dataPtr = sqlite3_column_text(stmt, 2) else {
                continue
            }
            items.append(EncryptedItem(
                id: String(cString: idPtr),
                type: String(cString: typePtr),
                encryptedDataBase64: String(cString: dataPtr)
            ))
        }

        return items
    }

    /// Resolve the database path in the App Group shared container.
    private static func getDatabasePath() throws -> String {
        guard let containerURL = FileManager.default.containerURL(
            forSecurityApplicationGroupIdentifier: "group.com.keykeykey.shared"
        ) else {
            throw DatabaseError.notFound("App Group container not available")
        }

        let dbPath = containerURL.appendingPathComponent("keykeykey.db").path

        guard FileManager.default.fileExists(atPath: dbPath) else {
            throw DatabaseError.notFound(dbPath)
        }

        return dbPath
    }
}
```

```swift
/// Compares the last two segments of the hostname (registrable domain).
/// Does not handle multi-level TLDs (.co.uk) — acceptable limitation
/// since app identifier matching is prioritized.
struct DomainMatcher {

    /// Extract the registrable domain (last two segments) from a URL string.
    /// e.g., "https://login.github.com/oauth" → "github.com"
    static func extractRegistrableDomain(_ urlString: String) -> String? {
        guard let url = URL(string: urlString),
              let host = url.host?.lowercased() else { return nil }
        let parts = host.split(separator: ".")
        guard parts.count >= 2 else { return String(host) }
        return parts.suffix(2).joined(separator: ".")
    }

    /// Check if a credential matches by app identifier (case-insensitive).
    static func matchesByAppIdentifier(
        credential appIdentifiers: [String],
        query appIdentifier: String
    ) -> Bool {
        let lower = appIdentifier.lowercased()
        return appIdentifiers.contains { $0.lowercased() == lower }
    }

    /// Check if a credential matches by domain.
    static func matchesByDomain(
        credentialURL: String?,
        queryDomain: String
    ) -> Bool {
        guard let url = credentialURL else { return false }
        guard let credDomain = extractRegistrableDomain(url) else { return false }
        guard let queryRegDomain = extractRegistrableDomain(
            queryDomain.contains("://") ? queryDomain : "https://\(queryDomain)"
        ) else { return false }
        return credDomain == queryRegDomain
    }
}
```

```swift
/// Decrypts all credential items, filters by match, zeros non-matching data.
static func findCredentials(
    appIdentifier: String?,
    domain: String?,
    dek: Data
) -> [MatchedCredential] {
    // Read encrypted items from shared SQLite
    let encryptedItems: [DatabaseReader.EncryptedItem]
    do {
        encryptedItems = try DatabaseReader.readCredentials()
    } catch {
        print("DatabaseReader error: \(error.localizedDescription)")
        return []
    }

    var results: [MatchedCredential] = []

    for item in encryptedItems {
        // Base64 decode
        guard let encryptedData = Data(base64Encoded: item.encryptedDataBase64) else {
            continue
        }

        // Decrypt
        let decryptedData: Data
        do {
            decryptedData = try CryptoBridge.decrypt(encryptedData, key: dek)
        } catch {
            continue // Skip items that fail decryption
        }

        // Parse JSON
        guard let json = try? JSONSerialization.jsonObject(with: decryptedData) as? [String: Any],
              let name = json["name"] as? String,
              let username = json["username"] as? String,
              let password = json["password"] as? String else {
            continue
        }

        let url = json["url"] as? String
        let appIds = json["appIdentifiers"] as? [String] ?? []

        // Check match
        var matched = false
        if let appId = appIdentifier, !appId.isEmpty {
            matched = DomainMatcher.matchesByAppIdentifier(
                credential: appIds, query: appId
            )
        }
        if !matched, let dom = domain, !dom.isEmpty {
            matched = DomainMatcher.matchesByDomain(credentialURL: url, queryDomain: dom)
        }

        if matched {
            results.append(MatchedCredential(
                id: item.id,
                name: name,
                username: username,
                password: password,
                url: url,
                appIdentifiers: appIds
            ))
        } else {
            // Zero decrypted data for non-matching credentials
            var mutableData = decryptedData
            mutableData.resetBytes(in: 0..<mutableData.count)
        }
    }

    return results
}
```

```swift
    // Read PIN data from shared Keychain
    guard let pinDataRaw = KeychainHelper.read(key: KeychainHelper.pinDataKey),
          let pinJson = try? JSONSerialization.jsonObject(with: pinDataRaw) as? [String: String],
          let wrappedDEKBase64 = pinJson["wrappedDEK"],

    // Check remaining attempts
    let attemptsRemaining = readPinAttempts() ?? 5
    if attemptsRemaining <= 0 {
        // Delete PIN data and show error
        KeychainHelper.delete(key: KeychainHelper.pinDataKey)
        deletePinAttempts()
        showUnsupportedAlert("Too many failed PIN attempts. Please unlock with the main app and re-enable PIN.")
        return
    }

    // Derive KEK from PIN via Argon2id (mobile preset, p=1)
    let params = Argon2Params(t: 2, m: 19_456, p: 1, dkLen: 32)
    let kek: Data
    do {

    } catch {
        showUnsupportedAlert("Key derivation failed: \(error.localizedDescription)")
        return
    }

    // Attempt to unwrap DEK
    do {
        let unwrappedDEK = try CryptoBridge.unwrapDEK(wrappedDEK, kek: kek)
        // Success — reset attempts and proceed
        savePinAttempts(5)

        // Zero KEK

        mutableKek.resetBytes(in: 0..<mutableKek.count)

            deletePinAttempts()
            showUnsupportedAlert("Too many failed PIN attempts. PIN has been disabled. Please unlock with the main app.")
```
